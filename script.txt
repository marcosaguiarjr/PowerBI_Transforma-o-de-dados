/* ============================================================
   1. CRIAÇÃO DO BANCO
   ============================================================ */

DROP DATABASE IF EXISTS azure_company;

CREATE DATABASE azure_company;

USE azure_company;


/* ============================================================
   2. TABELA DEPARTAMENT
   ============================================================ */

CREATE TABLE departament (
    Dname VARCHAR(15) NOT NULL,
    Dnumber INT NOT NULL,
    Mgr_ssn CHAR(9),
    Mgr_start_date DATE,
    Dept_create_date DATE,

    PRIMARY KEY (Dnumber),

    UNIQUE (Dname),

    CHECK (Dept_create_date < Mgr_start_date)
);


/* ============================================================
   3. TABELA EMPLOYEE
   ============================================================ */

CREATE TABLE employee (
    Fname VARCHAR(15) NOT NULL,
    Minit CHAR(1),
    Lname VARCHAR(15) NOT NULL,
    Ssn CHAR(9) NOT NULL,
    Bdate DATE,
    Address VARCHAR(30),
    Sex CHAR(1),
    Salary DECIMAL(10,2),
    Super_ssn CHAR(9),
    Dno INT NOT NULL DEFAULT 1,

    PRIMARY KEY (Ssn)
);


/* ============================================================
   4. FOREIGN KEYS DE EMPLOYEE
   ============================================================ */

ALTER TABLE employee
ADD FOREIGN KEY (Super_ssn)
REFERENCES employee(Ssn)
ON DELETE SET NULL
ON UPDATE CASCADE;


/* ============================================================
   5. FOREIGN KEY EMPLOYEE -> DEPARTAMENT
   ============================================================ */

ALTER TABLE employee
ADD FOREIGN KEY (Dno)
REFERENCES departament(Dnumber)
ON UPDATE CASCADE;


/* ============================================================
   6. FOREIGN KEY DEPARTAMENT -> EMPLOYEE
   ============================================================ */

ALTER TABLE departament
ADD FOREIGN KEY (Mgr_ssn)
REFERENCES employee(Ssn)
ON UPDATE CASCADE;


/* ============================================================
   7. TABELA DEPT_LOCATIONS
   ============================================================ */

CREATE TABLE dept_locations (
    Dnumber INT NOT NULL,
    Dlocation VARCHAR(15) NOT NULL,

    PRIMARY KEY (Dnumber, Dlocation)
);


/* ============================================================
   8. FOREIGN KEY DEPT_LOCATIONS
   ============================================================ */

ALTER TABLE dept_locations
ADD FOREIGN KEY (Dnumber)
REFERENCES departament(Dnumber)
ON DELETE CASCADE
ON UPDATE CASCADE;


/* ============================================================
   9. TABELA PROJECT
   ============================================================ */

CREATE TABLE project (
    Pname VARCHAR(15) NOT NULL,
    Pnumber INT NOT NULL,
    Plocation VARCHAR(15),
    Dnum INT NOT NULL,

    PRIMARY KEY (Pnumber),

    UNIQUE (Pname)
);


/* ============================================================
   10. FOREIGN KEY PROJECT -> DEPARTAMENT
   ============================================================ */

ALTER TABLE project
ADD FOREIGN KEY (Dnum)
REFERENCES departament(Dnumber)
ON UPDATE CASCADE;


/* ============================================================
   11. TABELA WORKS_ON
   ============================================================ */

CREATE TABLE works_on (
    Essn CHAR(9) NOT NULL,
    Pno INT NOT NULL,
    Hours DECIMAL(3,1) NOT NULL,

    PRIMARY KEY (Essn, Pno)
);


/* ============================================================
   12. FOREIGN KEYS WORKS_ON
   ============================================================ */

ALTER TABLE works_on
ADD FOREIGN KEY (Essn)
REFERENCES employee(Ssn)
ON UPDATE CASCADE;

ALTER TABLE works_on
ADD FOREIGN KEY (Pno)
REFERENCES project(Pnumber)
ON UPDATE CASCADE;


/* ============================================================
   13. TABELA DEPENDENT
   ============================================================ */

CREATE TABLE dependent (
    Essn CHAR(9) NOT NULL,
    Dependent_name VARCHAR(15) NOT NULL,
    Sex CHAR(1),
    Bdate DATE,
    Relationship VARCHAR(8),

    PRIMARY KEY (Essn, Dependent_name)
);


/* ============================================================
   14. FOREIGN KEY DEPENDENT -> EMPLOYEE
   ============================================================ */

ALTER TABLE dependent
ADD FOREIGN KEY (Essn)
REFERENCES employee(Ssn)
ON DELETE CASCADE
ON UPDATE CASCADE;

/* ============================================================
   23. DATA DE NASCIMENTO E ENDEREÇO DO JOHN SMITH
   ============================================================ */

SELECT Bdate, Address
FROM employee
WHERE Fname = 'John'
  AND Minit = 'B'
  AND Lname = 'Smith';


/* ============================================================
   24. DEPARTAMENTO RESEARCH
   ============================================================ */

SELECT *
FROM departament
WHERE Dname = 'Research';


/* ============================================================
   25. FUNCIONÁRIOS DO DEPARTAMENTO RESEARCH
   ============================================================ */

SELECT e.Fname, e.Lname, e.Address
FROM employee e
JOIN departament d
    ON e.Dno = d.Dnumber
WHERE d.Dname = 'Research';


/* ============================================================
   26. DEPARTAMENTOS E SUAS LOCALIZAÇÕES
   ============================================================ */

SELECT
    d.Dname AS Department,
    l.Dlocation AS Location
FROM departament d
JOIN dept_locations l
    ON d.Dnumber = l.Dnumber;


/* ============================================================
   27. DEPARTAMENTOS E SEUS GERENTES
   ============================================================ */

SELECT
    d.Dname AS Department,
    CONCAT(e.Fname, ' ', e.Lname) AS Manager
FROM departament d
JOIN employee e
    ON d.Mgr_ssn = e.Ssn;


/* ============================================================
   28. PROJETOS LOCALIZADOS EM STAFFORD
   ============================================================ */

SELECT *
FROM project
WHERE Plocation = 'Stafford';


/* ============================================================
   29. PROJETOS DE STAFFORD E SEUS GERENTES
   ============================================================ */

SELECT
    p.Pnumber,
    p.Dnum,
    e.Lname,
    e.Address,
    e.Bdate
FROM project p
JOIN departament d
    ON p.Dnum = d.Dnumber
JOIN employee e
    ON d.Mgr_ssn = e.Ssn
WHERE p.Plocation = 'Stafford';


/* ============================================================
   30. FUNCIONÁRIOS DOS DEPARTAMENTOS 3, 6 E 9
   ============================================================ */

SELECT *
FROM employee
WHERE Dno IN (3, 6, 9);


/* ============================================================
   31. SALÁRIO E INSS
   ============================================================ */

SELECT
    Fname,
    Lname,
    Salary,
    Salary * 0.011 AS INSS
FROM employee;


/* ============================================================
   32. INSS ARREDONDADO
   ============================================================ */

SELECT
    Fname,
    Lname,
    Salary,
    ROUND(Salary * 0.011, 2) AS INSS
FROM employee;


/* ============================================================
   33. EMPREGADOS QUE TRABALHAM NO PRODUCTX
   ============================================================ */

SELECT
    e.Fname,
    e.Lname,
    e.Salary * 1.10 AS increased_sal
FROM employee e
JOIN works_on w
    ON e.Ssn = w.Essn
JOIN project p
    ON w.Pno = p.Pnumber
WHERE p.Pname = 'ProductX';


/* ============================================================
   34. GERENTES E DEPARTAMENTOS
   ============================================================ */

SELECT
    d.Dname AS Department,
    CONCAT(e.Fname, ' ', e.Lname) AS Manager
FROM departament d
JOIN employee e
    ON d.Mgr_ssn = e.Ssn;


/* ============================================================
   35. FUNCIONÁRIOS DO RESEARCH
   ============================================================ */

SELECT
    e.Fname,
    e.Lname,
    e.Address
FROM employee e
JOIN departament d
    ON e.Dno = d.Dnumber
WHERE d.Dname = 'Research';


/* ============================================================
   36. FUNCIONÁRIOS E SEUS DEPENDENTES
   ============================================================ */

SELECT
    e.Ssn,
    CONCAT(e.Fname, ' ', e.Lname) AS Employee,
    COUNT(d.Essn) AS quantidade_dependentes
FROM employee e
LEFT JOIN dependent d
    ON e.Ssn = d.Essn
GROUP BY e.Ssn, e.Fname, e.Lname;


/* ============================================================
   37. VERIFICAÇÃO FINAL
   ============================================================ */

SELECT
    (SELECT COUNT(*) FROM employee) AS employee,
    (SELECT COUNT(*) FROM departament) AS departament,
    (SELECT COUNT(*) FROM dept_locations) AS dept_locations,
    (SELECT COUNT(*) FROM project) AS project,
    (SELECT COUNT(*) FROM works_on) AS works_on,
    (SELECT COUNT(*) FROM dependent) AS dependent;
